import java.net.*;
import java.io.*;

public class ChatServer {
    public static void main(String[] args) throws Exception {
        ServerSocket ss = new ServerSocket(2050);
        System.out.println("Server is waiting...");
        Socket soc = ss.accept();
        System.out.println("Client connected.");

        OutputStream os = soc.getOutputStream();
        System.out.println("Message to client:");
        int c = 0;

        while (c != '#') {
            c = System.in.read();
            os.write((byte)c);
        }

        InputStream is = soc.getInputStream();
        System.out.println("\nMessage from client:");
        int d = 0;

        while (d != '#') {
            d = is.read();
            System.out.print((char)d);
        }

        is.close();
        os.close();
        soc.close();
        ss.close();
    }
}

2. Echo server: 
  

import java.net.*;
import java.io.*;

public class EchoServer {
    public static void main(String[] args) throws Exception {
        ServerSocket ss = new ServerSocket(2050);
        System.out.println("Echo Server is waiting...");
        Socket s = ss.accept();
        System.out.println("Client connected.");

        InputStream is = s.getInputStream();
        OutputStream os = s.getOutputStream();

        int data;
        System.out.println("Message from client:");

        while ((data = is.read()) != -1) {
            System.out.print((char)data);
            os.write(data);
            if (data == '#')
                break;
        }

        is.close();
        os.close();
        s.close();
        ss.close();
    }
}

import java.net.*;

        InputStream is = s.getInputStream();
        OutputStream os = s.getOutputStream();

        System.out.println("Enter message to send:");
        int c = 0;

        while (c != '#') {
            c = System.in.read();
            os.write((byte)c);
        }

        System.out.println("\nEcho message from server:");
        int d = 0;

        while (d != '#') {
            d = is.read();
            System.out.print((char)d);
        }

        is.close();
        os.close();
        s.close();
    }
}


3.Time server: 


import java.net.*;
import java.io.*;
import java.time.LocalDateTime;

public class TimeServer {
    public static void main(String[] args) throws Exception {
        ServerSocket ss = new ServerSocket(2052);
        System.out.println("Time Server is waiting...");

        Socket soc = ss.accept();
        System.out.println("Client connected.");

        OutputStream os = soc.getOutputStream();

        String time = "Current Date and Time: "
                      + LocalDateTime.now() + "#";

        os.write(time.getBytes());

        System.out.println("Time sent to client.");

        os.close();
        soc.close();
        ss.close();
    }
}

import java.net.*;
import java.io.*;

public class TimeClient {
    public static void main(String[] args) throws Exception {
        Socket soc = new Socket("localhost", 2052);

        InputStream is = soc.getInputStream();
        System.out.println("Message from server:");

        int c;
        while ((c = is.read()) != -1 && c != '#') {
            System.out.print((char)c);
        }

        is.close();
        soc.close();
    }
